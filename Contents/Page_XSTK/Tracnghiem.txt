<script>
    // Cấu hình đáp án đúng và thời gian làm bài (tính bằng giây)
    const correctAnswers = {
        q1: "C",
        q2: "B",
        q3: "A"
    };
    let timeLeft = 30; // 30 giây làm bài
    let timerInterval;
    let isQuizFinished = false;

    // Hàm bắt đầu làm bài
    function startQuiz() {
        document.getElementById('start-btn').style.display = 'none'; // Ẩn nút bắt đầu
        document.getElementById('timer').style.display = 'block';   // Hiện đồng hồ
        document.getElementById('quiz-section').style.display = 'block'; // Hiện câu hỏi

        // Cập nhật đồng hồ hiển thị ban đầu
        updateTimerDisplay();

        // Chạy đếm ngược mỗi 1 giây
        timerInterval = setInterval(function() {
            timeLeft--;
            updateTimerDisplay();

            if (timeLeft <= 0) {
                clearInterval(timerInterval);
                submitQuiz(true); // Tự động nộp bài khi hết giờ (truyền tham số true)
            }
        }, 1000);
    }

    // Hàm cập nhật hiển thị thời gian định dạng mm:ss
    function updateTimerDisplay() {
        let minutes = Math.floor(timeLeft / 60);
        let seconds = timeLeft % 60;
        
        // Thêm số 0 phía trước nếu số có 1 chữ số
        minutes = minutes < 10 ? '0' + minutes : minutes;
        seconds = seconds < 10 ? '0' + seconds : seconds;

        document.getElementById('timer').textContent = `Thời gian: ${minutes}:${seconds}`;
    }

    // Hàm nộp bài và tính điểm
    function submitQuiz(isTimeout = false) {
        if (isQuizFinished) return; // Nếu đã nộp rồi thì không xử lý lại
        isQuizFinished = true;

        // Dừng đồng hồ đếm ngược
        clearInterval(timerInterval);

        // Vô hiệu hóa (khóa) tất cả các ô chọn để người dùng không sửa được nữa
        const inputs = document.querySelectorAll('input[type="radio"]');
        inputs.forEach(input => input.disabled = true);
        document.getElementById('submit-btn').disabled = true;

        // Tính điểm
        let score = 0;
        const totalQuestions = Object.keys(correctAnswers).length;
        const formData = new FormData(document.getElementById('quiz-form'));

        for (let key in correctAnswers) {
            if (formData.get(key) === correctAnswers[key]) {
                score++;
            }
        }

        // Hiển thị kết quả công bố điểm
        document.getElementById('score-text').textContent = `Số câu đúng: ${score}/${totalQuestions} (Đạt ${(score/totalQuestions*10).toFixed(1)} điểm)`;
        
        if (isTimeout) {
            document.getElementById('time-status').textContent = "Bài tập đã tự động ngắt vì HẾT GIỜ!";
            document.getElementById('time-status').style.color = "red";
        } else {
            document.getElementById('time-status').textContent = "Bạn đã hoàn thành bài tập chủ động.";
            document.getElementById('time-status').style.color = "green";
        }

        document.getElementById('result').style.display = 'block';
        
        // Cuộn màn hình xuống vùng kết quả để người dùng dễ thấy
        document.getElementById('result').scrollIntoView({ behavior: 'smooth' });
    }
</script>
